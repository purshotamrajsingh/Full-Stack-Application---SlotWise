   465→function Kpi({
   466→  icon: Icon,
   467→  label,
   468→  value,
   469→  sub,
   470→  tone,
   471→  delta,
   472→  loading,
   473→}: {
   474→  icon: typeof Banknote;
   475→  label: string;
   476→  value?: string;
   477→  sub?: string;
   478→  tone?: "good" | "bad" | "neutral";
   479→  /** Previous-period comparison pill. */
   480→  delta?: {
   481→    pct: number | null;
   482→    prev: string;
   483→    label: string;
   484→    /** For metrics where a decrease is good (e.g. no-show rate). */
   485→    goodWhenDown?: boolean;
   486→  };
   487→  loading?: boolean;
   488→}) {
   489→  const deltaTone =
   490→    delta?.pct == null
   491→      ? null
   492→      : delta.pct === 0
   493→        ? "flat"
   494→        : (delta.pct > 0) !== (delta.goodWhenDown ?? false)
   495→          ? "up"
   496→          : "down";
   497→
   498→  return (
   499→    <Card className="py-0 transition-all duration-200 hover:shadow-md hover:ring-1 hover:ring-primary/15">
   500→      <CardContent className="p-4">
   501→        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   502→          <Icon className="size-3.5" />
   503→          {label}
   504→        </div>
   505→        {loading || value === undefined ? (
   506→          <Skeleton className="mt-2 h-8 w-24" />
   507→        ) : (
   508→          <p
   509→            className={cn(
   510→              "mt-1 text-2xl font-bold leading-tight tabular-nums",
   511→              tone === "good" && "text-emerald-700 dark:text-emerald-400",
   512→              tone === "bad" && "text-rose-700 dark:text-rose-400",
   513→            )}
   514→          >
   515→            {value}
   516→          </p>
   517→        )}
   518→        {sub && !loading && (
   519→          <p className="mt-0.5 flex items-center gap-1 text-xs text-muted-foreground">
   520→            {tone === "good" && <ArrowUpRight className="size-3" />}
   521→            {tone === "bad" && <ArrowDownRight className="size-3" />}
   522→            {sub}
   523→          </p>
   524→        )}
   525→        {delta && !loading && (
   526→          <div className="mt-2 flex items-center gap-1.5">
   527→            <span
   528→              className={cn(
   529→                "inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[10px] font-semibold tabular-nums",
   530→                deltaTone === "up" &&
   531→                  "bg-emerald-100 text-emerald-800 dark:bg-emerald-950/60 dark:text-emerald-300",
   532→                deltaTone === "down" &&
   533→                  "bg-rose-100 text-rose-800 dark:bg-rose-950/60 dark:text-rose-300",
   534→                (deltaTone === "flat" || deltaTone === null) &&
   535→                  "bg-muted text-muted-foreground",
   536→              )}
   537→              title={`${delta.label}: was ${delta.prev}`}
   538→            >
   539→              {deltaTone === "up" && <ArrowUpRight className="size-3" />}
   540→              {deltaTone === "down" && <ArrowDownRight className="size-3" />}
   541→              {delta.pct == null ? "n/a" : `${delta.pct > 0 ? "+" : ""}${delta.pct}%`}
   542→            </span>
   543→            <span className="text-[10px] text-muted-foreground">{delta.label}</span>
   544→          </div>
   545→        )}
   546→      </CardContent>
   547→    </Card>
   548→  );
   549→}

   758→/**
   759→ * Admin broadcast composer — one message, every bell. Audience choices:
   760→ * everyone, customers only, or just customers with upcoming bookings.
   761→ * Each recipient gets their own outbox row + realtime push.
   762→ */
   763→function BroadcastDialog({
   764→  open,
   765→  onOpenChange,
   766→}: {
   767→  open: boolean;
   768→  onOpenChange: (v: boolean) => void;
   769→}) {
   770→  const [audience, setAudience] = useState<"ALL" | "CUSTOMERS" | "UPCOMING">("CUSTOMERS");
   771→  const [title, setTitle] = useState("");
   772→  const [body, setBody] = useState("");
   773→  const [sending, setSending] = useState(false);
   774→
   775→  const audienceHint: Record<string, string> = {
   776→    ALL: "Every account — owners, staff and customers.",
   777→    CUSTOMERS: "Customer accounts only — the usual choice for promos and notices.",
   778→    UPCOMING: "Only customers with a pending or confirmed future booking.",
   779→  };
   780→
   781→  async function send(e: React.FormEvent) {
   782→    e.preventDefault();
   783→    if (title.trim().length < 3 || body.trim().length < 5) {
   784→      toast.error("Give the message a title (3+ chars) and a body (5+ chars).");
   785→      return;
   786→    }
   787→    setSending(true);
   788→    try {
   789→      const res = await apiFetch<{ sent: number }>("/api/admin/broadcast", {
   790→        method: "POST",
   791→        body: { title: title.trim(), body: body.trim(), audience },
   792→      });
   793→      toast.success("Broadcast sent.", {
   794→        description: `${res.sent} recipient${res.sent === 1 ? "" : "s"} got the message in their bell.`,
   795→      });
   796→      setTitle("");
   797→      setBody("");
   798→      setAudience("CUSTOMERS");
   799→      onOpenChange(false);
   800→    } catch (err) {
   801→      toast.error(err instanceof Error ? err.message : "Could not send the broadcast.");
   802→    } finally {
   803→      setSending(false);
   804→    }
   805→  }
   806→
   807→  return (
   808→    <Dialog open={open} onOpenChange={onOpenChange}>
   809→      <DialogContent className="sm:max-w-md">
   810→        <DialogHeader>
   811→          <DialogTitle className="flex items-center gap-2">
   812→            <Megaphone className="size-5 text-primary" />
   813→            Send a broadcast
   814→          </DialogTitle>
   815→          <DialogDescription>
   816→            Lands instantly in every recipient&apos;s notification bell —
   817→            and in the mock email outbox.
   818→          </DialogDescription>
   819→        </DialogHeader>
   820→
   821→        <form className="flex flex-col gap-4" onSubmit={send}>
   822→          <div className="flex flex-col gap-1.5">
   823→            <Label>Audience</Label>
   824→            <Select value={audience} onValueChange={(v) => setAudience(v as typeof audience)}>
   825→              <SelectTrigger aria-label="Broadcast audience">
   826→                <SelectValue />
   827→              </SelectTrigger>
   828→              <SelectContent>
   829→                <SelectItem value="ALL">Everyone</SelectItem>
   830→                <SelectItem value="CUSTOMERS">All customers</SelectItem>
   831→                <SelectItem value="UPCOMING">Customers with upcoming bookings</SelectItem>
   832→              </SelectContent>
   833→            </Select>
   834→            <p className="text-xs text-muted-foreground">{audienceHint[audience]}</p>
   835→          </div>
   836→
   837→          <div className="flex flex-col gap-1.5">
   838→            <div className="flex items-center justify-between">
   839→              <Label htmlFor="bc-title">Title</Label>
   840→              <span className="text-xs tabular-nums text-muted-foreground">
   841→                {title.length}/80
   842→              </span>
   843→            </div>
   844→            <Input
   845→              id="bc-title"
   846→              required
   847→              minLength={3}
   848→              maxLength={80}
   849→              value={title}
   850→              onChange={(e) => setTitle(e.target.value)}
   851→              placeholder="October offer — 20% off hygiene visits"
   852→            />
   853→          </div>
   854→
   855→          <div className="flex flex-col gap-1.5">
   856→            <div className="flex items-center justify-between">
   857→              <Label htmlFor="bc-body">Message</Label>
   858→              <span className="text-xs tabular-nums text-muted-foreground">
   859→                {body.length}/500
   860→              </span>
   861→            </div>
   862→            <Textarea
   863→              id="bc-body"
   864→              required
   865→              minLength={5}
   866→              maxLength={500}
   867→              value={body}
   868→              onChange={(e) => setBody(e.target.value)}
   869→              placeholder="Mention the promo, when it runs and how to book…"
   870→              className="min-h-28"
   871→            />
   872→          </div>
   873→
   874→          <DialogFooter>
   875→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   876→              Cancel
   877→            </Button>
   878→            <Button type="submit" className="press" disabled={sending}>
   879→              {sending && <Loader2 className="size-4 animate-spin" />}
   880→              <Megaphone className="size-4" />
   881→              Send broadcast
   882→            </Button>
   883→          </DialogFooter>
   884→        </form>
   885→      </DialogContent>
   886→    </Dialog>
   887→  );
   888→}
